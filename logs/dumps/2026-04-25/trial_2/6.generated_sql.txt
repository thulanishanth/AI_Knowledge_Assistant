SELECT MAX(lead_time) AS highest_lead_time FROM hotel_reservations WHERE booking_status = 'Canceled' AND no_of_special_requests = 2 AND lead_time >= 0 AND lead_time <= 443 AND arrival_date >= 1 AND arrival_date <= 31 AND arrival_month >= 1 AND arrival_month <= 12 AND no_of_weekend_nights >= 0 AND no_of_weekend_nights <= 7 LIMIT 11;
